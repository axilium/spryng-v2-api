## API overview

This endpoint allows you to update or change the URL for the application which receives the event data from your webhook subscription. You can configure a different URL per event type if desired. 
Note: the receiving service must be a publicly accessible web page or web service which you host.

## URL & method

**Endpoint URL**

```
https://api.spryng.nl/v2/webhooks/events/{eventType}
```

**Method: PUT**

## Headers & input parameters

These parameters are mandatory to be provided in the header of your request. If not provided, or incorrect, your request will fail.

| Header | Type | Description |
| --- | --- | --- |
| Content-Type | String | application/json |
| X-Api-Key | String | API Key value for authorization |
| AccountReference | String | Your Spryng account reference - e.g. SPNL0000000 |

**Required input path parameters**

The below parameters are required within the path.

| Parameter | Type | Description |
| --- | --- | --- |
| eventType | String | The eventId of the event where you'd like to change the receiving URL |

**Example request header**

```

-H 'Api-Key: YOUR_API_KEY'
```

```java
import java.io.OutputStream;
import java.net.HttpURLConnection;
import java.net.URI;
import java.net.URL;
import java.net.URLEncoder;
import java.nio.charset.StandardCharsets;
import java.util.HashMap;

public class Main {
    public static void main(String[] args) throws Exception {
        String baseUrl = "https://api.spryng.nl/v2/webhooks/events/%7BeventType%7D";

        URL url = URI.create(baseUrl).toURL();
        HttpURLConnection connection = (HttpURLConnection) url.openConnection();
        connection.setRequestMethod("PUT");

        connection.setRequestProperty("AccountReference", "your-AccountReference-header-here");
        
        connection.setRequestProperty("X-Api-Key", "YOUR_KEY_HERE");

        String body = """
        [
  {
    "url": "url",
    "platform": "platform",
    "optionalNotes": "optionalNotes"
  }
]""";

        connection.setDoOutput(true);

        try (OutputStream outputStream = connection.getOutputStream()) {
            byte[] input = body.getBytes(StandardCharsets.UTF_8);
            outputStream.write(input, 0, input.length);
        }
        catch (Exception e) {
            System.err.println("Failed due to error: " + e.getMessage());
        }
        
        int responseCode = connection.getResponseCode();

        if (responseCode >= 200 && responseCode < 300) {
          try (var reader = new java.io.BufferedReader(
            new java.io.InputStreamReader(connection.getInputStream(), StandardCharsets.UTF_8))) {
              StringBuilder response = new StringBuilder();
              String line;

              while ((line = reader.readLine()) != null) {
                  response.append(line.trim());
              }
            }
        }
        else {
          try (var errorReader = new java.io.BufferedReader(new java.io.InputStreamReader(connection.getErrorStream(), StandardCharsets.UTF_8))) {
            StringBuilder errorResponse = new StringBuilder();
            String line;

            while ((line = errorReader.readLine()) != null) {
                errorResponse.append(line.trim());
            }

            System.err.println("Error Response: " + errorResponse.toString());
          } catch (Exception e) {
            System.err.println("Failed to read error response: " + e.getMessage());
          }
        }

        connection.disconnect();
    }
}
```
